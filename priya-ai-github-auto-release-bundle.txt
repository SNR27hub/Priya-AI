  push:
    branches: [ main, master ]
  workflow_dispatch:

  contents: write
  actions: write
  packages: write

jobs:
  build:
    name: Build Priya AI APK
    runs-on: ubuntu-latest

    steps:
      - name: 1. Checkout Repository
        uses: actions/checkout@v4

      - name: 2. Set up Java 17
        uses: actions/setup-java@v4
        with:
          distribution: 'temurin'
          java-version: '17'

      - name: 3. Set up Gradle CLI
        uses: gradle/actions/setup-gradle@v3

      - name: 4. Set up Android SDK
        uses: android-actions/setup-android@v3

      - name: 5. Accept Android Licenses
        run: yes | sdkmanager --licenses || true

      - name: 6. Compile Android APK
        run: |
          BUILD_DIR="."
          if [ -f "./android/build.gradle.kts" ]; then
            BUILD_DIR="./android"
          fi
          echo "Building Android project in $BUILD_DIR"
          cd "$BUILD_DIR"

          if [ ! -f "./gradlew" ] || [ ! -f "./gradle/wrapper/gradle-wrapper.jar" ]; then
            gradle wrapper --gradle-version 8.2 || true
          fi

          chmod +x ./gradlew || true
          ./gradlew assembleDebug --stacktrace || gradle assembleDebug --stacktrace || ./gradlew assembleRelease --stacktrace || gradle assembleRelease --stacktrace

      - name: 7. Find and Rename Output APK
        run: |
          mkdir -p output-apk
          APK_PATH=$(find . -name "*.apk" -type f | head -n 1)
          if [ -n "$APK_PATH" ]; then
            echo "Successfully found APK at: $APK_PATH"
            cp "$APK_PATH" output-apk/PriyaAI-v1.0-release.apk
            ls -lh output-apk/
          else
            echo "Searching for APK in subdirectories..."
            find . -name "*.apk"
          fi

      - name: 8. Upload APK to GitHub Artifacts (Direct Download)
        uses: actions/upload-artifact@v4
        with:
          name: PriyaAI-v1.0-APK
          path: output-apk/PriyaAI-v1.0-release.apk
          if-no-files-found: warn

      - name: 9. Publish Official GitHub Release
        continue-on-error: true
        uses: softprops/action-gh-release@v2
        with:
          tag_name: "v1.0.${{ github.run_number }}"

package com.snr.priyaai.services

import android.app.*
import android.content.*
import android.os.*
import android.speech.tts.TextToSpeech
import androidx.core.app.NotificationCompat
import java.util.Locale

class PriyaBackgroundVoiceService : Service(), TextToSpeech.OnInitListener {

    private lateinit var tts: TextToSpeech
    private var isTtsReady = false
    private val CHANNEL_ID = "PriyaAIVoiceChannel"

    private val batteryReceiver = object : BroadcastReceiver() {
        override fun onReceive(context: Context?, intent: Intent?) {
            when (intent?.action) {
                Intent.ACTION_POWER_CONNECTED -> {
                    speak("Aha! Charger lag gaya hai. Phone ab charge ho raha hai!")
                }
                Intent.ACTION_POWER_DISCONNECTED -> {
                    val batteryIntent = registerReceiver(null, IntentFilter(Intent.ACTION_BATTERY_CHANGED))
                    val level = batteryIntent?.getIntExtra(BatteryManager.EXTRA_LEVEL, -1) ?: -1
                    speak("Charger hata diya gaya hai. Battery abhi $level percent bachi hai.")
                }
                Intent.ACTION_BATTERY_LOW -> {
                    speak("Warning! Battery bahut kam ho gayi hai. Kripya phone ko charge me lagaiye.")
                }
            }
        }
    }

    override fun onCreate() {
        super.onCreate()
        tts = TextToSpeech(this, this)
        createNotificationChannel()

        val notification: Notification = NotificationCompat.Builder(this, CHANNEL_ID)
            .setContentTitle("Priya AI Active")
            .setContentText("Listening for 'Wake up Priya' | Battery alerts enabled")
            .setSmallIcon(android.R.drawable.ic_btn_speak_now)
            .setOngoing(true)
            .build()

        startForeground(101, notification)

        val filter = IntentFilter().apply {
            addAction(Intent.ACTION_POWER_CONNECTED)
            addAction(Intent.ACTION_POWER_DISCONNECTED)
            addAction(Intent.ACTION_BATTERY_LOW)
        }
        registerReceiver(batteryReceiver, filter)
    }

    override fun onInit(status: Int) {
        if (status == TextToSpeech.SUCCESS) {
            val result = tts.setLanguage(Locale("hi", "IN"))
            if (result == TextToSpeech.LANG_MISSING_DATA || result == TextToSpeech.LANG_NOT_SUPPORTED) {
                tts.setLanguage(Locale.ENGLISH)
            }
            tts.setPitch(1.15f)
            tts.setSpeechRate(0.96f)
            isTtsReady = true
        }
    }

    fun speak(text: String) {
        if (isTtsReady) {
            tts.speak(text, TextToSpeech.QUEUE_FLUSH, null, "PriyaTTS")
        }
    }

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int = START_STICKY

    override fun onDestroy() {
        unregisterReceiver(batteryReceiver)
        tts.stop()
        tts.shutdown()
        super.onDestroy()
    }

    override fun onBind(intent: Intent?): IBinder? = null

    private fun createNotificationChannel() {
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            val channel = NotificationChannel(
                CHANNEL_ID,
                "Priya AI Assistant",
                NotificationManager.IMPORTANCE_LOW
            )
            val manager = getSystemService(NotificationManager::class.java)
            manager?.createNotificationChannel(channel)
        }
    }
}

package com.snr.priyaai.services

import android.accessibilityservice.AccessibilityService
import android.accessibilityservice.GestureDescription
import android.graphics.Path
import android.graphics.PixelFormat
import android.view.*
import android.view.accessibility.AccessibilityEvent
import android.view.accessibility.AccessibilityNodeInfo
import android.widget.ImageView

class PriyaAccessibilityService : AccessibilityService() {

    private var windowManager: WindowManager? = null
    private var floatingView: View? = null

    override fun onServiceConnected() {
        super.onServiceConnected()
        showFloatingBubble()
    }

    private fun showFloatingBubble() {
        windowManager = getSystemService(WINDOW_SERVICE) as WindowManager
        val params = WindowManager.LayoutParams(
            WindowManager.LayoutParams.WRAP_CONTENT,
            WindowManager.LayoutParams.WRAP_CONTENT,
            WindowManager.LayoutParams.TYPE_APPLICATION_OVERLAY,
            WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE,
            PixelFormat.TRANSLUCENT
        ).apply {
            gravity = Gravity.TOP or Gravity.START
            x = 50
            y = 200
        }

        val imageView = ImageView(this).apply {
            setImageResource(android.R.drawable.ic_dialog_info)
        }
        floatingView = imageView
        try {
            windowManager?.addView(floatingView, params)
        } catch (e: Exception) {
            e.printStackTrace()
        }
    }

    fun sendWhatsAppMessage() {
        val root = rootInActiveWindow ?: return
        val sendNodes = root.findAccessibilityNodeInfosByViewId("com.whatsapp:id/send")
        if (sendNodes.isNotEmpty()) {
            sendNodes[0].performAction(AccessibilityNodeInfo.ACTION_CLICK)
        }
    }

    override fun onAccessibilityEvent(event: AccessibilityEvent?) {}

    override fun onInterrupt() {}

    override fun onDestroy() {
        if (floatingView != null) {
            try {
                windowManager?.removeView(floatingView)
            } catch (e: Exception) {}
        }
        super.onDestroy()
    }
}

package com.snr.priyaai.services

import android.content.Intent
import android.service.notification.NotificationListenerService
import android.service.notification.StatusBarNotification

class PriyaNotificationListener : NotificationListenerService() {

    override fun onNotificationPosted(sbn: StatusBarNotification?) {
        val packageName = sbn?.packageName ?: return
        val extras = sbn.notification.extras
        val title = extras.getString("android.title") ?: ""
        val text = extras.getCharSequence("android.text")?.toString() ?: ""

        if (packageName.contains("whatsapp") || packageName.contains("dialer") || packageName.contains("telecom")) {
            val intent = Intent("com.snr.priyaai.NEW_NOTIFICATION").apply {
                putExtra("package", packageName)
                putExtra("title", title)
                putExtra("text", text)
            }
            sendBroadcast(intent)
        }
    }

    override fun onNotificationRemoved(sbn: StatusBarNotification?) {}
}
